SQL 
SELECT region AS "区域",
  SUM(revenue) AS "总营收",
  ROUND((SUM(revenue) - LAG(SUM(revenue)) OVER (PARTITION BY region ORDER BY quarter))
    / LAG(SUM(revenue)) OVER (...) * 100, 1) AS "增长率"
FROM sales_data WHERE quarter = 'Q3' AND year = 2024
GROUP BY region ORDER BY "总营收" DESC;
